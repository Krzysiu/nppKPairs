cmake_minimum_required(VERSION 3.20)
project(KPairs VERSION 0.2.0 LANGUAGES CXX RC)
option(KPAIRS_INTEGRATION_TEST "Run isolated host tests on startup" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(KPairs SHARED src/plugin.cpp src/plugin.rc)
target_include_directories(KPairs PRIVATE vendor src tests "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(KPairs PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00 WIN32_LEAN_AND_MEAN)
if(KPAIRS_INTEGRATION_TEST)
  target_compile_definitions(KPairs PRIVATE KPAIRS_INTEGRATION_TEST)
endif()
target_link_libraries(KPairs PRIVATE comctl32 comdlg32 shell32 ole32 uuid)
set_target_properties(KPairs PROPERTIES PREFIX "")
if(MINGW)
  target_compile_options(KPairs PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Wno-misleading-indentation)
  target_link_options(KPairs PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(model_tests tests/model_tests.cpp)
  target_include_directories(model_tests PRIVATE vendor src)
  target_compile_definitions(model_tests PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(model_tests PRIVATE ole32)
  if(MINGW)
    target_link_options(model_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME model_tests COMMAND model_tests)
endif()
